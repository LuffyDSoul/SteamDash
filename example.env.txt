# Copia este archivo como .env y reemplaza los valores CHANGE_ME.
# Nunca publiques el archivo .env real: contiene credenciales.

# Spring Boot carga automaticamente el .env del directorio del servicio cuando
# la aplicacion lo importa. Para el arranque local, usa estas variables en el
# entorno del proceso o copia el .env en dacs-be/dacs-conector.

# -----------------------------------------------------------------------------
# APIs externas usadas por dacs-conector
# -----------------------------------------------------------------------------

# Steam Web API. Requerida para bibliotecas, perfiles, noticias y logros.
# Obtener en: https://steamcommunity.com/dev/apikey
STEAM_API_KEY=CHANGE_ME_STEAM_API_KEY
STEAM_WEB_API_URL=https://api.steampowered.com

# Steam Store API. No requiere API key.

# Cliente Feign generico del conector.

# SteamSpy se consulta publicamente desde el codigo y no requiere API key.

STEAMSPY_API_URL=https://steamspy.com/api.php

# -----------------------------------------------------------------------------
# Comunicacion interna
# -----------------------------------------------------------------------------

# Frontend -> BFF
FRONTEND_BFF_URL=http://localhost:9001/bff

# BFF -> conector. Incluye el context path /conector.
CONECTOR_URL=http://localhost:9002/conector

# BFF -> backend. El context path /backend se agrega en application.yml.
MS_BACKEND_URL=http://localhost:9003

# -----------------------------------------------------------------------------
# PostgreSQL usado por dacs-backend
# -----------------------------------------------------------------------------

DB_URL=jdbc:postgresql://localhost:5432/dacs_db
DB_USER=dacs_user
DB_PASSWORD=CHANGE_ME_DATABASE_PASSWORD

# -----------------------------------------------------------------------------
# Keycloak usado por dacs-fe
# -----------------------------------------------------------------------------
# El cliente es publico y usa PKCE; no debe existir CLIENT_SECRET en el
# frontend. Angular no lee .env directamente: copia estos valores en los
# archivos src/environments/environment*.ts.
KEYCLOAK_URL=http://localhost:8080
KEYCLOAK_REALM=dacs
KEYCLOAK_CLIENT_ID=dacs-fe
KEYCLOAK_CLIENT_ID_PRODUCTION=dacs-app

# -----------------------------------------------------------------------------
# Puertos locales de referencia
# -----------------------------------------------------------------------------

BFF_PORT=9001
CONECTOR_PORT=9002
BACKEND_PORT=9003
KEYCLOAK_PORT=8080
